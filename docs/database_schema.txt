Table transaksi {
  nomor text [not null, note: 'Text identifier for transaction']

  bukti text
  nama_pasien text
  keterangan text
  status text [default: 'diajukan', note: 'diajukan | selesai']
  total numeric [default: 0]
  processed_by uuid
  processed_at timestamptz
  created_at timestamptz [default: `now()`]
  created_by uuid
  updated_at timestamptz [default: `now()`]
  updated_by uuid
  deleted_at timestamptz
}

Table transaksi_detail {
  id bigint [pk, not null]

  batch text
  harga numeric
  subtotal numeric

Table transaksi_jasa {
  id bigint [pk, not null]
